//Show Listing Route MVC
module.exports.showListing = async (req, res) => {
  let {id} = req.params;
  const listing = await Listing.findById(id)
    .populate({
      path: "reviews",
      populate: {
        path: "author",
      },
    }) //reviews ka info + har review ka author ka info bhi (nested populate)
    .populate("owner");

  if(!listing) {
    req.flash("error", "Listing you requested for does not exist");
    return res.redirect("/listings");
  }

  res.render("listings/show.ejs", {listing});

};


//New Listing Route MVC
module.exports.createListing = async (req, res, next) => {
  const newListing = new Listing(req.body.listing);
  newListing.owner = req.user._id;
  if (req.file) {
    newListing.image = { url: req.file.path, filename: req.file.filename };
  }

  //location se coordinates nikalke save karo
  const geometry = await geocode(req.body.listing.location, req.body.listing.country);
  if (geometry) {
    newListing.geometry = geometry;
  }

  await newListing.save();
  req.flash("success", "New Listing Created!");
  res.redirect("/listings");
};
//Edit Route MVC
module.exports.renderEditForm = async (req, res) => {
  let { id } = req.params;
  const listing = await Listing.findById(id);
  if(!listing) {
    req.flash("error", "Listing you requested for does not exist");
    return res.redirect("/listings");
  }
  res.render("listings/edit.ejs", { listing });
};


//Update Listing Route MVC
module.exports.updateListing = async (req, res) => {
  let { id } = req.params;
  let listing = await Listing.findByIdAndUpdate(id, { ...req.body.listing });

  const geometry = await geocode(req.body.listing.location, req.body.listing.country);
  if (geometry) {
    listing.geometry = geometry;
  }
  if (req.file) {
    listing.image = { url: req.file.path, filename: req.file.filename };
  }
  await listing.save();

  req.flash("success", "Listing Updated!");
  res.redirect(`/listings/${id}`);
};


//Delete Listing Route MVC
module.exports.deleteListing = async (req, res) => {
  let { id } = req.params;
  let deletedListing = await Listing.findByIdAndDelete(id);
  console.log(deletedListing);
  req.flash("success", " Listing Deleted!");
  res.redirect("/listings");
}; 
===== models\listing.js ===== 
const mongoose = require("mongoose");
const Schema = mongoose.Schema;
const Review = require("./review.js");

const listingSchema = new Schema({
  title: {
    type: String,
    required: true,
  },

  description: String,

  image: {
    filename: String,
    url: {
      type: String,
      default:
        "https://images.unsplash.com/photo-1518780664697-55e3ad937233?w=800",
    },
  },

  price: Number,
  location: String,
  country: String,
  reviews: [
    {
      type: Schema.Types.ObjectId, //Har Review document ka MongoDB automatically ek _id banata hai., Phir Listing ke reviews array mein usi Review ki _id store hoti hai:
      ref: "Review",
    },
  ],
  owner: {
    type: Schema.Types.ObjectId,
    ref: "User",
  },
  //map ke liye coordinates (GeoJSON format), coordinates: [longitude, latitude]
  geometry: {
    type: {
      type: String,
      enum: ["Point"],
    },
    coordinates: {
      type: [Number],
    },
  },
});


//jab hamra delete route listing ka trigger hoga to ye middlwear chlega 
listingSchema.post("findOneAndDelete", async (listing) => {
  if (listing) {
    await Review.deleteMany({ _id: {$in: listing.reviews } });
  }
});
//Mongoose ka rule hai: schema middleware hamesha us model ki file me likha jaata hai jiske schema pe wo attach ho raha hai.




const Listing = mongoose.model("Listing", listingSchema);

module.exports = Listing; 
===== schema.js ===== 
//joi package used
const Joi = require("joi");

module.exports.listingSchema = Joi.object({
    listing: Joi.object({
        title: Joi.string().required(),
        description: Joi.string().required(),
        location: Joi.string().required(),
        country: Joi.string().required(),
        price: Joi.number().required(),
        //image ab string (URL) ya object ({filename, url}) dono allow karta hai, kyunki edit form se object aa sakta hai
        image: Joi.alternatives().try(
            Joi.string().allow("", null),
            Joi.object({
                filename: Joi.string().allow("", null),
                url: Joi.string().allow("", null),
            })
        ),
    }).required(),
});

module.exports.reviewSchema = Joi.object({
    review: Joi.object({
        rating: Joi.number().required().min(1).max(5),
        comment: Joi.string().required(),
    }).required(),
});
